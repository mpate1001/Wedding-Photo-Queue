**3. Change initClient() to return a Promise<Client>:**
Change the function signature to `async function initClient(): Promise<Client>`.

Instead of calling `client.initialize().catch(...)` fire-and-forget, wrap the initialization in a promise that resolves when the client emits 'ready':

```typescript
async function initClient(): Promise<Client> {
  const executablePath = process.env.PUPPETEER_EXECUTABLE_PATH;
  if (!executablePath) {
    const msg = '[WhatsApp] PUPPETEER_EXECUTABLE_PATH is not set. On the VPS use /usr/bin/google-chrome-stable. Aborting WhatsApp initialization.';
    console.error(msg);
    throw new Error(msg);
  }

  const client = new Client({
    authStrategy: new LocalAuth({ dataPath: '.wwebjs_auth' }),
    puppeteer: {
      executablePath,
      args: ['--no-sandbox', '--disable-setuid-sandbox'],
    },
  });

  global.__whatsappStatus = 'initializing';
  global.__whatsappQR = undefined;

  // Keep all existing event handlers (qr, ready, auth_failure, disconnected)
  client.on('qr', (qr) => {
    global.__whatsappStatus = 'qr_pending';
    global.__whatsappQR = qr;
    console.log('[WhatsApp] QR code ready — scan below with your phone:');
    qrcode.generate(qr, { small: true });
  });

  client.on('auth_failure', (msg) => {
    global.__whatsappStatus = 'auth_failure';
    console.error('[WhatsApp] Auth failure:', msg);
  });

  client.on('disconnected', (reason) => {
    global.__whatsappStatus = 'disconnected';
    global.__whatsappClient = undefined;
    global.__whatsappInitPromise = undefined;  // Allow re-initialization after disconnect
    console.warn('[WhatsApp] Disconnected:', reason);
  });

  // Return a promise that resolves when client is ready
  return new Promise<Client>((resolve, reject) => {
    client.on('ready', () => {
      global.__whatsappStatus = 'ready';
      global.__whatsappQR = undefined;
      global.__whatsappClient = client;
      console.log('[WhatsApp] Client ready');
      resolve(client);
    });

    console.log('[WhatsApp] Initializing with Chromium at:', executablePath);
    client.initialize().catch((err) => {
      console.error('[WhatsApp] Initialization error:', err);
      global.__whatsappStatus = 'disconnected';
      global.__whatsappInitPromise = undefined;  // Allow retry after failure
      reject(err);
    });
  });
}
```

**4. Add whenReady() export — the primary API for callers:**
```typescript
export async function whenReady(): Promise<Client> {
  // If client is already ready, return immediately
  if (global.__whatsappClient && global.__whatsappStatus === 'ready') {
    return global.__whatsappClient;
  }

  // If initialization is in progress, wait for the same promise (no duplicate init)
  if (!global.__whatsappInitPromise) {
    global.__whatsappInitPromise = initClient();
  }

  return global.__whatsappInitPromise;
}
```
